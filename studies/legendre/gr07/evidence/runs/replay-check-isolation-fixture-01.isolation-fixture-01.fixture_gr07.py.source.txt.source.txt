import os,json,subprocess
from pathlib import Path
p=Path(os.environ['GR07_PROBE_SOURCE'])
operations=[lambda:p.write_text('changed'),lambda:p.chmod(0o777),lambda:p.rename(p.with_name('renamed')),lambda:p.unlink()]
blocked=[]
for operation in operations:
    try:operation()
    except OSError as e:blocked.append(e.errno)
assert len(blocked)==4,blocked
assert not Path('/run/user/1000/bus').exists()
# Detached descendant must disappear with this namespace, even after leader exit.
subprocess.Popen(['/usr/bin/setsid','/usr/bin/sleep','20'],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
Path('/var/tmp/fixture-result.json').write_text(json.dumps({'blocked_operations':blocked,'pid_namespace':os.readlink('/proc/self/ns/pid'),'detached_descendant_spawned':True}))
print('PASS: source writes/chmod/rename/unlink denied; service-manager sockets masked; detached-child cleanup checked by controller.')
