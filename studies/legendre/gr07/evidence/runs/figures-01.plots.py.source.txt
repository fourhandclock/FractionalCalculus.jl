"""Standalone scientific SVG figures from retained CSVs; standard library only.

Both axes use explicit log10 transforms. SVG is an exportable vector artifact;
no external fonts, plotting package, browser or network resource is required.
No figure is made from invalid_initial_results.csv or Python timing values.
"""
import csv,math,sys,json,statistics,hashlib,os
from pathlib import Path
from html import escape

COLORS=['#0072b2','#d55e00','#009e73','#cc79a7','#e69f00','#56b4e9','#333333']
VARIANTS=['L1-printed','L12-printed','L12-PI-L','Alikhanov-PI-L','Alikhanov-PI-Q']


def read(p):
    with Path(p).open() as f:return list(csv.DictReader(f))


def panel(series,title,xlabel,ylabel,width=580,height=400,bars=()):
    points=[(float(x),float(y)) for name,rows,color,dash in series for x,y in rows if float(x)>0 and float(y)>0]
    points += [(xlo,y) for x,y,xlo,xhi,color in bars]+[(xhi,y) for x,y,xlo,xhi,color in bars]
    assert points, 'no positive data to plot'
    xs=[math.log10(x) for x,y in points];ys=[math.log10(y) for x,y in points]
    lo,hi=min(xs),max(xs);low,high=min(ys),max(ys)
    if hi==lo:lo-=.2;hi+=.2
    if high==low:low-=.2;high+=.2
    pad=.06*(high-low);low-=pad;high+=pad
    left,top,right,bottom=82,43,20,77
    W,H=width-left-right,height-top-bottom
    X=lambda x:left+(math.log10(x)-lo)/(hi-lo)*W
    Y=lambda y:top+H-(math.log10(y)-low)/(high-low)*H
    lines=[f'<svg xmlns="http://www.w3.org/2000/svg" width="{width}" height="{height}" viewBox="0 0 {width} {height}">',
           '<rect width="100%" height="100%" fill="white"/>',
           f'<text x="{width/2}" y="23" text-anchor="middle" font-family="sans-serif" font-size="15">{escape(title)}</text>']
    for axis,lower,upper in [('x',lo,hi),('y',low,high)]:
        for power in range(math.floor(lower),math.ceil(upper)+1):
            if not lower<=power<=upper:continue
            if axis=='x':
                x=X(10**power)
                lines += [f'<path d="M{x},{top} V{top+H}" stroke="#dddddd"/>',f'<text x="{x}" y="{top+H+19}" text-anchor="middle" font-size="11">10^{power}</text>']
            else:
                y=Y(10**power)
                lines += [f'<path d="M{left},{y} H{left+W}" stroke="#dddddd"/>',f'<text x="{left-9}" y="{y+4}" text-anchor="end" font-size="11">10^{power}</text>']
    lines += [f'<path d="M{left},{top} V{top+H} H{left+W}" fill="none" stroke="black"/>',
              f'<text x="{left+W/2}" y="{height-15}" text-anchor="middle" font-size="13">{escape(xlabel)} (log scale)</text>',
              f'<text transform="translate(18 {top+H/2}) rotate(-90)" text-anchor="middle" font-size="13">{escape(ylabel)} (log scale)</text>']
    for x,y,xlo,xhi,color in bars:
        yy=Y(y); xl=X(xlo); xr=X(xhi)
        lines.append(f'<path d="M{xl},{yy} H{xr} M{xl},{yy-3} V{yy+3} M{xr},{yy-3} V{yy+3}" fill="none" stroke="{color}" stroke-width="1.3"/>')
    for i,(name,rows,color,dash) in enumerate(series):
        rows=[(float(x),float(y)) for x,y in rows if float(x)>0 and float(y)>0]
        coords=' '.join(f'{X(x):.4f},{Y(y):.4f}' for x,y in sorted(rows))
        lines.append(f'<polyline points="{coords}" fill="none" stroke="{color}" stroke-width="1.8" stroke-dasharray="{dash}"/>')
        for x,y in rows:lines.append(f'<circle cx="{X(x)}" cy="{Y(y)}" r="2.7" fill="{color}"/>')
        yy=top+13+15*i
        lines += [f'<path d="M{left+9},{yy-4} h20" stroke="{color}" stroke-width="2" stroke-dasharray="{dash}"/>',
                  f'<text x="{left+35}" y="{yy}" font-size="10" font-family="sans-serif">{escape(name)}</text>']
    lines.append('</svg>')
    return '\n'.join(lines)


def write(path,text):
    with path.open('x') as f:f.write(text+'\n')


def main(results,timings,destination):
    rows=read(results);perf=read(timings);out=Path(destination);out.mkdir(parents=True,exist_ok=True)
    created=[]
    for a,b in [(.5,.25),(.75,.5),(.9,.8)]:
        for kind in ['smooth','singular']:
            series=[]
            for i,v in enumerate(VARIANTS):
                matches=[r for r in rows if float(r['alpha'])==a and float(r['beta'])==b and r['case']==kind and r['variant']==v and int(r['M'])==14]
                for grading in sorted({float(r['grading']) for r in matches}):
                    data=[(r['N'],r['h1_max']) for r in matches if float(r['grading'])==grading]
                    series.append((v+(' graded' if grading>1 else ''),data,COLORS[i],'6,3' if grading>1 else ''))
            name=f'error_N_a{a:g}_{kind}.svg';write(out/name,panel(series,f'α={a:g}, β={b:g}, {kind}; M=14','N','max continuous H1 error'));created.append(name)
    for kind in ['smooth','singular']:
        series=[]; bars=[]
        for i,v in enumerate(VARIANTS):
            data=[r for r in perf if r['phase']=='total' and r['variant']==v and r['case']==kind]
            points=[]
            for N in sorted({int(r['N']) for r in data}):
                samples=[r for r in data if int(r['N'])==N]
                median=statistics.median(float(r['seconds']) for r in samples); error=statistics.median(float(r['h1_max']) for r in samples)
                points.append((median,error)); bars.append((median,error,min(float(r['seconds']) for r in samples),max(float(r['seconds']) for r in samples),COLORS[i]))
            if points:series.append((v,points,COLORS[i],''))
        name=f'error_runtime_{kind}.svg';write(out/name,panel(series,f'Julia warm totals; α=.75, β=.5, {kind}','median seconds (bars: min–max)','max continuous H1 error',bars=bars));created.append(name)
    series=[]
    for i,v in enumerate(VARIANTS):
        data=[r for r in perf if r['phase']=='total' and r['case']=='smooth' and r['variant']==v]
        points=sorted({(int(r['N']),int(r['time_weight_bytes'])+int(r['history_bytes'])) for r in data})
        series.append((v,points,COLORS[i],''))
    name='stored_arrays_N.svg';write(out/name,panel(series,'Stored A/W/state-control arrays + solution history','N','stored array bytes'));created.append(name)
    series=[]
    for i,v in enumerate(VARIANTS[-2:]):
        data=[(r['M'],r['h1_max']) for r in rows if r['variant']==v and r['case']=='singular' and float(r['alpha'])==.75 and int(r['N'])==256 and int(r['M']) in [3,5,7,9,12,16]]
        series.append((v,data,COLORS[i+3],''))
    name='spatial_refinement.svg';write(out/name,panel(series,'Graded singular case; α=.75, β=.5, N=256','M','max continuous H1 error'));created.append(name)
    with (out/'figure-data.json').open('x') as f:json.dump({'path_base':'GR07 study','scientific_csv':os.path.relpath(results,Path(__file__).resolve().parents[1]),'scientific_sha256':hashlib.sha256(Path(results).read_bytes()).hexdigest(),'timing_csv':os.path.relpath(timings,Path(__file__).resolve().parents[1]),'timing_sha256':hashlib.sha256(Path(timings).read_bytes()).hexdigest(),'figures':created,'axes':'Explicit log10 coordinates; no fitted or extrapolated curves','invalid_initial_input_used':False},f,indent=2)
    print('Created',len(created),'standalone SVG figures from actual Julia data only.')

if __name__=='__main__':main(*sys.argv[1:])
